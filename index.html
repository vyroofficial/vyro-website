package com.example.freetv.data import com.example.freetv.parser.EpgGuide import com.example.freetv.parser.EpgParser import kotlinx.coroutines.flow.MutableStateFlow import kotlinx.coroutines.flow.StateFlow import kotlinx.coroutines.flow.asStateFlow import java.net.URL class EpgRepository { private val _guide = MutableStateFlow(null) val guide: StateFlow = _guide.asStateFlow() private val _lastError = MutableStateFlow(null) val lastError: StateFlow = _lastError.asStateFlow() fun loadFromUrl(url: String) { try { val connection = URL(url).openConnection() connection.connectTimeout = 10_000 connection.readTimeout = 15_000 connection.getInputStream().use { input -> _guide.value = EpgParser.parse(input) _lastError.value = null } } catch (ex: Exception) { _lastError.value = ex.message ?: "Failed to load EPG" } } fun clear() { _guide.value = null } }
